#!/bin/bash
opt_log=$1
aws s3 cp \
   --recursive \
   s3://mrituyanjay2104/${opt_log}/ \
   /home/ubuntu/s3-logs/templog/ \
   --profile default


tmp_file_location=/home/ubuntu/s3-logs/templog/*
base_location=/home/ubuntu/s3-logs/logs


# Create new directory to store today's messages:
today=$(date +"%m_%d_%Y")
[[ -d ${base_location}/"$today" ]] || mkdir ${base_location}/"$today"

# Give the message files readable names:
for FILE in $tmp_file_location
do
   mv $FILE ${base_location}/${today}/
   zip -r ${today}.zip $FILE ${base_location}/${today}/
done
